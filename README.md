## Python Program:
```
import os
from openai import OpenAI
from google import genai
import anthropic

# --------------------------------------------------
# 1. Initialize API Clients
# --------------------------------------------------

openai_client = OpenAI(
    api_key=os.getenv("OPENAI_API_KEY")
)

gemini_client = genai.Client(
    api_key=os.getenv("GEMINI_API_KEY")
)

claude_client = anthropic.Anthropic(
    api_key=os.getenv("ANTHROPIC_API_KEY")
)

# --------------------------------------------------
# 2. Common Prompt
# --------------------------------------------------

prompt = """
Suggest five practical ways for a college student
to improve productivity using AI tools.

For each suggestion:
1. Give the idea.
2. Explain it briefly.
3. Mention one practical benefit.
"""

# --------------------------------------------------
# 3. OpenAI Response
# --------------------------------------------------

def get_openai_response(prompt):
    response = openai_client.responses.create(
        model="gpt-5-mini",
        input=prompt
    )

    return response.output_text


# --------------------------------------------------
# 4. Gemini Response
# --------------------------------------------------

def get_gemini_response(prompt):
    response = gemini_client.models.generate_content(
        model="gemini-2.5-flash",
        contents=prompt
    )

    return response.text


# --------------------------------------------------
# 5. Claude Response
# --------------------------------------------------

def get_claude_response(prompt):
    response = claude_client.messages.create(
        model="claude-sonnet-4-5",
        max_tokens=1000,
        messages=[
            {
                "role": "user",
                "content": prompt
            }
        ]
    )

    return response.content[0].text


# --------------------------------------------------
# 6. Call Multiple AI Tools
# --------------------------------------------------

responses = {}

try:
    responses["OpenAI"] = get_openai_response(prompt)
except Exception as e:
    responses["OpenAI"] = "API error: " + str(e)

try:
    responses["Gemini"] = get_gemini_response(prompt)
except Exception as e:
    responses["Gemini"] = "API error: " + str(e)

try:
    responses["Claude"] = get_claude_response(prompt)
except Exception as e:
    responses["Claude"] = "API error: " + str(e)


# --------------------------------------------------
# 7. Display Responses
# --------------------------------------------------

print("=" * 70)
print("MULTIPLE AI TOOL OUTPUT")
print("=" * 70)

for tool, response in responses.items():

    print("\n" + "-" * 70)
    print(tool)
    print("-" * 70)

    print(response)


# --------------------------------------------------
# 8. Basic Comparison
# --------------------------------------------------

print("\n" + "=" * 70)
print("COMPARISON")
print("=" * 70)

for tool, response in responses.items():

    word_count = len(response.split())

    print(f"\n{tool}")
    print("Word Count:", word_count)

    if "API error:" in response:
        print("Status: API request failed")
    else:
        print("Status: Response received successfully")


# --------------------------------------------------
# 9. Generate Actionable Insights
# --------------------------------------------------

print("\n" + "=" * 70)
print("ACTIONABLE INSIGHTS")
print("=" * 70)

print("""
1. Compare the common recommendations provided by all AI tools.

2. Identify recommendations that appear in only one AI response.

3. Prefer recommendations that are practical, clear,
   and relevant to the student's requirements.

4. Combine the strongest recommendations from the
   different AI responses into a final action plan.

5. Verify important information before implementing it.
""")
```
